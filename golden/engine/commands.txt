# Commands (ADR 0016, docs/spec/conversation.md section 6): a message whose
# first character is `/` names an intent directly. Both languages' names
# work; the reply is in the conversation's language; the rest of the line
# fills the declared slot through the same extractor as a captured slot, so
# the outcome is the sentence form's.

[es] U: /calc 1 + 2
     T: 1 + 2 = 3.                                → answered · calc.arithmetic · expression=1+2

# `/calc` reads the rest of the line as the sentence form would (the
# command is declared `sentence = true`): a percentage or a conversion
# reaches its own intent with what that intent's rule captured.
[es] U: /calc 15% de 80
     T: El 15 % de 80 es 12.                      → answered · calc.percent · amount=80 percent=15

[es] U: /calc 5 km a millas
     T: 5 km son ≈ 3,11 mi.                       → answered · calc.convert · amount=5 from=km to=mi

# The conversation's language holds, but a line its rules do not read is
# read by the other language's.
[es] U: /calc 15% of 80
     T: El 15 % de 80 es 12.                      → answered · calc.percent · amount=80 percent=15

Given: hint = en
[en] U: /calc 1 + 2
     T: 1 + 2 = 3.                                → answered · calc.arithmetic · expression=1+2

[en] U: /calc 15% of 80
     T: 15% of 80 is 12.                          → answered · calc.percent · amount=80 percent=15

[en] U: /calc 5 km in miles
     T: 5 km is ≈ 3.11 mi.                        → answered · calc.convert · amount=5 from=km to=mi

# A Spanish name in an English conversation answers in English.
Given: draw = 4
[en] U: /dado
     T: It came up 4.                             → answered · random.dice · count=1 sides=6

Given: hint = none
Given: draw = option 2
[es] U: /pick pizza, sushi o tacos
     T: Elijo sushi.                              → answered · random.pick · options=pizza|sushi|tacos

[es] U: /falta navidad
     T: Faltan 286 días para Navidad (sábado, 25 de diciembre de 2027).
                                                  → answered · clock.until · target=2027-12-25

# The conversation's language holds: a command line is no evidence.
[en] U: what time is it
     T: It's 18:42.                               → answered · clock.now
[en] U: /hora
     T: It's 18:42.                               → answered · clock.now

# A missing argument is a missing slot: asked, and the next message answers.
[es] U: /calc
     T: ¿Qué quieres calcular?                    → asks · calc.arithmetic · expression
[es] U: 6 por 7
     T: 6 × 7 = 42.                               → answered · calc.arithmetic · expression=6*7

# Unknown commands: never the rules or the reflector; the reply names the
# command and how to list them.
[es] U: /clac 1 + 2
     T: No conozco el comando /clac. Escribe /ayuda para ver los que hay.
                                                  → unhandled · unknown-command

Given: hint = en
[en] U: /clac 1 + 2
     T: I don't know the command /clac. Type /help to see the ones there are.
                                                  → unhandled · unknown-command

# A command that takes no argument says so instead of ignoring it.
[en] U: /fortune please
     T: The command /fortune takes nothing after it.
                                                  → unhandled · fortune.tell

# The dice command takes the rest of the line as one free-text argument its
# handler reads (ADR 0016): notation, or a number of sides. The outcome is
# the sentence form's ("roll 2d20").
Given: draw = 3, 20
[en] U: /roll 2d20
     T: 3 and 20: total 23.                       → answered · random.dice · count=2 sides=20

Given: draw = 17
[en] U: /dado 20
     T: It came up 17.                            → answered · random.dice · count=1 sides=20

Given: draw = 5
[en] U: /roll d20
     T: It came up 5.                             → answered · random.dice · count=1 sides=20

Given: draw = 8, 1, 4
[en] U: /roll 3d8+2
     T: 8, 1 and 4, plus 2: total 15.             → answered · random.dice · count=3 modifier=2 sides=8

Given: draw = 6, 2
[en] U: /roll 2d6-1
     T: 6 and 2, minus 1: total 7.                → answered · random.dice · count=2 modifier=-1 sides=6

[en] U: /roll three dice
     T: Try: roll a die, 2d6 or 3d8+2.            → unhandled · random.dice

# A command closes an open question, as an unrelated message would.
Given: hint = none
[es] U: elige uno
     T: ¿Entre qué opciones?                      → asks · random.pick · options
[es] U: /calc 2 * 3
     T: 2 × 3 = 6.                                → answered · calc.arithmetic · expression=2*3

# /help and /ayuda list what is installed, with its commands.
Given: hint = en
[en] U: /ayuda
     T: Here's what I can do:
        · say hello and goodbye
        · say who and what I am
        · help you use what I can do (/help)
        · the time, the date and how long until something (/time, /date, /weekday, /until)
        · the time in other places and time differences (/timein)
        · arithmetic, percentages and unit conversion (/calc)
        · roll dice, flip coins, pick among options and give random numbers (/roll, /coin, /pick, /number)
        · give a saying or a fortune (/fortune)
        · tell you what a word means (/define)
        · learn your own phrases for what I can already do
        · translate words and set phrases between Spanish and English (/translate)
        · tell you what Wikipedia says about something or someone (/wiki)
        · remember your name, where you live and your units
                                                  → answered · help.capabilities

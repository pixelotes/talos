# Open questions (docs/spec/conversation.md section 3): Talos asks for a
# missing slot, and the next message of the conversation is offered to the
# question first. The spec's examples use skills not built yet (timezones,
# preferences); these use the built ones. Turns with no blank line between
# them are one conversation.

# 3.2 The next message answers it: bare, framed, or in either language. The
# outcome is the original intent, completed.
[es] U: elige uno
     T: ¿Entre qué opciones?                       → asks · random.pick · options
Given: draw = option 2
[es] U: pues entre té o café, por favor
     T: Elijo café.                               → answered · random.pick · options=té|café

Given: hint = en
[en] U: pick one
     T: Pick among what?                          → asks · random.pick · options
Given: draw = option 1
[en] U: tea or coffee please
     T: I pick tea.                               → answered · random.pick · options=tea|coffee

Given: hint = none
[es] U: calcula
     T: ¿Qué quieres calcular?                     → asks · calc.arithmetic · expression
[en] U: what is 7 times 6
     T: 7 × 6 = 42.                               → answered · calc.arithmetic · expression=7*6
     (the request repeated, in English: answered in English)

# 3.3 Cancelling: the question closes, nothing runs, no intent.
[es] U: elige uno
     T: ¿Entre qué opciones?                       → asks · random.pick · options
[es] U: déjalo
     T: Vale, lo dejo.                            → answered · cancelled
[es] U: pizza o sushi
     T: No sé qué responder a eso.                → unhandled · catch-all
     (the question was cancelled: "pizza o sushi" is not a request)

Given: hint = en
[en] U: pick one
     T: Pick among what?                          → asks · random.pick · options
[en] U: never mind
     T: Okay, dropped.                            → answered · cancelled

Given: hint = none
[es] U: cuánto falta
     T: ¿Para cuándo?                              → asks · clock.until · target
[es] U: no, déjalo, da igual
     T: Vale, lo dejo.                            → answered · cancelled

# With nothing open, a cancellation word is a bare acknowledgement.
[es] U: olvídalo
     T: Vale.                                     → answered · acknowledgement

# 3.4 Asking again: a choice when the answer is ambiguous; and never a third
# time.
Given: hint = en
Given: now = 2027-03-14 18:42 America/New_York
[en] U: how long until 3/4
     T: Do you mean 3 April or 4 March?           → asks · clock.until · target
[en] U: the first one
     T: 20 days to go (Saturday 3 April 2027).    → answered · clock.until · target=2027-04-03

[en] U: what weekday is 3/4/2027
     T: Do you mean 3 April or 4 March?           → asks · clock.weekday · date
[en] U: no
     T: I didn't catch that. Do you mean 3 April or 4 March?
                                                  → asks · clock.weekday · date
[en] U: maybe the other
     T: I couldn't work that out, so I'll leave it.
                                                  → unhandled · clock.weekday

Given: hint = none

# 3.7 Hesitations keep the question open, and count as turns: after three
# further turns it has expired.
Given: hint = none
[es] U: elige uno
     T: ¿Entre qué opciones?                       → asks · random.pick · options
[es] U: mmm
     T: Tómate tu tiempo.                         → answered · hedge
[es] U: a ver
     T: Vale.                                     → answered · hedge
[es] U: espera
     T: Aquí sigo.                                → answered · hedge
[es] U: pizza o sushi
     T: Eso no lo he pillado.                     → unhandled · catch-all
     (expired after three turns)
